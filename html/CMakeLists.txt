# This library has no editor, SDK, window-system or GPU dependency.
cmake_minimum_required(VERSION 3.25)
set(_html_test_default OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NemoDevHtml LANGUAGES C CXX)
    set(_html_test_default ON)
endif()
function(nemodev_html_dependencies)
    set(BUILD_SHARED_LIBS OFF)
    set(LITEHTML_BUILD_TESTING OFF)
    include("${CMAKE_CURRENT_SOURCE_DIR}/detail/Litehtml.cmake")
endfunction()
nemodev_html_dependencies()

add_library(NemoDevHtml STATIC HtmlDocument.cpp detail/HtmlPainting.cpp)
target_include_directories(NemoDevHtml PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(NemoDevHtml PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(NemoDevHtml PRIVATE /utf-8 /EHsc)
endif()
target_link_libraries(NemoDevHtml PRIVATE litehtml)
set_target_properties(NemoDevHtml PROPERTIES FOLDER "NemoDev/HTML")

option(NEMODEV_HTML_BUILD_TESTS "Build the standalone HTML renderer tests" ${_html_test_default})
if(NEMODEV_HTML_BUILD_TESTS)
    enable_testing()
    add_executable(NemoDevHtmlTests tests/HtmlDocumentTests.cpp)
    target_link_libraries(NemoDevHtmlTests PRIVATE NemoDevHtml)
    add_test(NAME NemoDevHtml COMMAND NemoDevHtmlTests)
endif()
